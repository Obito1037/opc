import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// Original, deliberately stylized fitting mannequin. All surfaces are generated
// locally. The articulated preview is not a reconstruction or a size guarantee.
const TAU = Math.PI * 2;
const clamp = THREE.MathUtils.clamp;
const mat = (color, options = {}) => new THREE.MeshStandardMaterial({
  color, roughness: 0.65, metalness: 0, ...options,
});

function mesh(parent, geometry, material, position = [0, 0, 0], scale) {
  const object = new THREE.Mesh(geometry, material);
  object.position.set(...position);
  if (scale) object.scale.set(...scale);
  object.castShadow = true;
  object.receiveShadow = true;
  parent.add(object);
  return object;
}

function group(parent, position = [0, 0, 0]) {
  const object = new THREE.Group();
  object.position.set(...position);
  parent.add(object);
  return object;
}

function oval(parent, material, position, scale, segments = 28) {
  return mesh(parent, new THREE.SphereGeometry(1, segments, 20), material, position, scale);
}

// Elliptical loft with Catmull-Rom interpolation makes continuous shoulder,
// waist, hip and garment contours, rather than stacks of primitives.
function loft(rings, segments = 48, subdivisions = 3, pleats = 0) {
  const contour = new THREE.CatmullRomCurve3(
    rings.map(([y, x, z]) => new THREE.Vector3(x, y, z)), false, 'catmullrom', 0.22,
  );
  const steps = (rings.length - 1) * subdivisions;
  const positions = [], uvs = [], indices = [];
  for (let j = 0; j <= steps; j++) {
    const p = contour.getPoint(j / steps);
    for (let i = 0; i <= segments; i++) {
      const angle = i / segments * TAU;
      const fold = 1 + pleats * Math.cos(angle * 14) * (1 - j / steps);
      positions.push(Math.sin(angle) * p.x * fold, p.y, Math.cos(angle) * p.z * fold);
      uvs.push(i / segments, j / steps);
      if (j < steps && i < segments) {
        const a = j * (segments + 1) + i, b = a + segments + 1;
        indices.push(a, a + 1, b, a + 1, b + 1, b);
      }
    }
  }
  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return geometry;
}

function line(parent, points, material, radius = 0.008, closed = false) {
  const curve = new THREE.CatmullRomCurve3(points.map(p => new THREE.Vector3(...p)), closed);
  return mesh(parent, new THREE.TubeGeometry(curve, Math.max(12, points.length * 5), radius, 6, closed), material);
}

function patch(parent, points, material) {
  const geometry = new THREE.BufferGeometry();
  const indices = [];
  for (let i = 1; i < points.length - 1; i++) indices.push(0, i, i + 1);
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(points.flat(), 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return mesh(parent, geometry, material);
}

function ellipseRing(parent, material, y, rx, rz, radius = 0.009) {
  const points = [];
  for (let i = 0; i < 40; i++) {
    const a = i / 40 * TAU;
    points.push([Math.sin(a) * rx, y, Math.cos(a) * rz]);
  }
  return line(parent, points, material, radius, true);
}

function fabricTexture() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 128;
  const ctx = canvas.getContext('2d');
  ctx.fillStyle = '#b8b8b8';
  ctx.fillRect(0, 0, 128, 128);
  for (let i = 0; i < 128; i += 4) {
    ctx.fillStyle = i % 8 ? '#b3b3b3' : '#c5c5c5';
    ctx.fillRect(i, 0, 1, 128);
    ctx.fillRect(0, i, 128, 1);
  }
  const texture = new THREE.CanvasTexture(canvas);
  texture.wrapS = texture.wrapT = THREE.RepeatWrapping;
  texture.repeat.set(5, 5);
  return texture;
}

function shadowTexture() {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 128;
  const ctx = canvas.getContext('2d');
  const gradient = ctx.createRadialGradient(64, 64, 4, 64, 64, 64);
  gradient.addColorStop(0, 'rgba(36,31,55,.29)');
  gradient.addColorStop(0.45, 'rgba(36,31,55,.13)');
  gradient.addColorStop(1, 'rgba(36,31,55,0)');
  ctx.fillStyle = gradient;
  ctx.fillRect(0, 0, 128, 128);
  return new THREE.CanvasTexture(canvas);
}

function hairShell() {
  const positions = [], indices = [];
  const around = 56, down = 24;
  for (let j = 0; j <= down; j++) {
    for (let i = 0; i <= around; i++) {
      const angle = i / around * TAU;
      // Open forehead and face, while retaining a full bob at the back.
      const end = 2.3 - 1.3 * Math.max(0, Math.cos(angle)) ** 4;
      const theta = (j / down) * end;
      positions.push(
        Math.sin(angle) * Math.sin(theta) * 0.216,
        Math.cos(theta) * 0.285 + 0.027,
        Math.cos(angle) * Math.sin(theta) * 0.203 - 0.022,
      );
      if (i < around && j < down) {
        const a = j * (around + 1) + i, b = a + around + 1;
        indices.push(a, b, a + 1, a + 1, b, b + 1);
      }
    }
  }
  const geometry = new THREE.BufferGeometry();
  geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  return geometry;
}

export class AvatarStage {

      container.appendChild(notice);

      console.warn('AvatarStage WebGL unavailable:', error);
      return;

    this.renderer.setClearColor(0x000000, 0);
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;

    canvas.style.cssText = 'width:100%;height:100%;display:block;touch-action:none;outline:none';

    this.lastTime = null;
    this.tick = this.tick.bind(this);
    this.frame = requestAnimationFrame(this.tick);
    this.contextLost = event => {
      event.preventDefault();

    canvas.addEventListener('webglcontextlost', this.contextLost);
  }

    key.position.set(-3, 6, 5);

    key.shadow.camera.left = key.shadow.camera.bottom = -3;
    key.shadow.camera.right = key.shadow.camera.top = 3;

    this.floor.castShadow = false;

      line(this.scene, [
        [Math.sin(angle) * 0.98, 0.002, Math.cos(angle) * 0.98],

  }

  setLook(update = {}) {
    this.look = { ...this.look, ...update };
    if (!this.avatar) return;
    const { garment, color, skin, shoe, accessory, makeup, body, showGarment, makeupIntensity } = this.look;
    const m = this.materials;
    if (color && /^#[\da-f]{3,8}$/i.test(color)) {
      m.fabric.color.set(color);
      m.lapel.color.copy(m.fabric.color).lerp(new THREE.Color('#ffffff'), 0.16);
    }
    if (skin && /^#[\da-f]{3,8}$/i.test(skin)) m.skin.color.set(skin);
    if (makeup && /^#[\da-f]{3,8}$/i.test(makeup)) {
      m.lips.color.set('#bb8177').lerp(new THREE.Color(makeup), clamp(Number(makeupIntensity) || 0, 0, 1));
    }
    const dressed = showGarment !== false;
    this.baseLayer.visible = !dressed;
    this.jacket.visible = dressed && garment === 'jacket';
    this.shirt.visible = dressed && garment === 'shirt';
    this.dress.visible = dressed && garment === 'dress';
    this.pantsWaist.visible = this.pantsSeat.visible = this.belt.visible = this.buckle.visible = !dressed || garment !== 'dress';
    for (const arm of this.arms) {
      arm.sleeve.visible = arm.sleeveLower.visible = arm.cuff.visible = arm.button.visible = dressed && garment === 'jacket';
      arm.shortSleeve.visible = dressed && (garment === 'shirt' || garment === 'dress');
    }
    for (const leg of this.legs) {
      leg.pantsTop.visible = leg.pantsBottom.visible = !dressed || garment !== 'dress';
      leg.sneakers.visible = shoe !== 'boots';
      leg.boots.visible = shoe === 'boots';
    }
    for (const [name, object] of Object.entries(this.accessories)) object.visible = name === accessory;
    // Body controls are illustrative proportions; never inferred measurements.
    let width = typeof body === 'number' ? body : body?.width ?? 1;
    let height = typeof body === 'object' ? body?.height ?? 1 : 1;
    if (typeof body === 'string') width = ({ slim: 0.91, regular: 1, standard: 1, curvy: 1.12 })[body] || 1;
    width = clamp(Number(width) || 1, 0.82, 1.24);
    height = clamp(Number(height) || 1, 0.9, 1.1);
    this.avatar.scale.set(width, height, 0.5 + width * 0.5);
    this.animatePose(this.elapsed, 0);
  }

  setMotion(motion) {
    this.motion = ['idle', 'walk', 'wave', 'turn'].includes(motion) ? motion : 'idle';
    this.elapsed = 0;
  }

  setPaused(paused) {
    this.paused = Boolean(paused);
  }

  setPhoto(image, landmarks = null) {
    if (!this.head || !image) return false;
    const imageWidth = image.naturalWidth || image.width;
    const imageHeight = image.naturalHeight || image.height;
    if (!imageWidth || !imageHeight) return false;
    // Face landmarks are normalized coordinates. Pose landmarks may also supply
    // a face region, while an unlocalized photo never becomes a fabricated face.
    const points = landmarks?.faceLandmarks?.[0] || landmarks?.landmarks?.[0] || landmarks;
    if (!Array.isArray(points) || points.length < 5) return false;
    const facePoints = points.length >= 400
      ? [points[10], points[152], points[234], points[454]]
      : points.slice(0, 11);
    if (facePoints.some(p => !p || !Number.isFinite(p.x) || !Number.isFinite(p.y))) return false;
    const xs = facePoints.map(p => p.x * imageWidth), ys = facePoints.map(p => p.y * imageHeight);
    let width = Math.max(...xs) - Math.min(...xs);
    let height = Math.max(...ys) - Math.min(...ys);
    if (width < 8 || height < 8) return false;
    const centerX = (Math.min(...xs) + Math.max(...xs)) / 2;
    const centerY = (Math.min(...ys) + Math.max(...ys)) / 2;
    if (points.length < 400) { width *= 1.24; height = Math.max(height * 1.75, width * 1.35); }
    else { width *= 1.08; height *= 1.06; }
    const canvas = document.createElement('canvas');
    canvas.width = 256; canvas.height = 320;
    const ctx = canvas.getContext('2d');
    ctx.drawImage(image, centerX - width / 2, centerY - height / 2, width, height, 0, 0, 256, 320);
    ctx.globalCompositeOperation = 'destination-in';
    const mask = ctx.createRadialGradient(128, 154, 56, 128, 154, 148);
    mask.addColorStop(0, 'rgba(255,255,255,1)');
    mask.addColorStop(0.62, 'rgba(255,255,255,1)');
    mask.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = mask;
    ctx.fillRect(0, 0, 256, 320);
    const texture = new THREE.CanvasTexture(canvas);
    texture.colorSpace = THREE.SRGBColorSpace;
    if (this.photoFace) {
      this.photoFace.material.map.dispose();
      this.photoFace.material.map = texture;
      this.photoFace.material.needsUpdate = true;
    } else {
      const positions = [], uvs = [], indices = [];
      const nx = 24, ny = 30;
      for (let j = 0; j <= ny; j++) {
        for (let i = 0; i <= nx; i++) {
          const x = (i / nx - 0.5) * 0.36;
          const y = (0.5 - j / ny) * 0.44 - 0.017;
          const z = 0.192 * Math.sqrt(Math.max(0.04, 1 - (x / 0.205) ** 2 - (y / 0.285) ** 2)) + 0.018;
          positions.push(x, y, z);
          uvs.push(i / nx, 1 - j / ny);
          if (i < nx && j < ny) {
            const a = j * (nx + 1) + i, b = a + nx + 1;
            indices.push(a, b, a + 1, a + 1, b, b + 1);
          }
        }
      }
      const geometry = new THREE.BufferGeometry();
      geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
      geometry.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
      geometry.setIndex(indices);
      geometry.computeVertexNormals();
      this.photoFace = mesh(this.head, geometry, mat('#ffffff', {
        map: texture, transparent: true, roughness: 0.9, depthWrite: false,
      }));
      this.photoFace.castShadow = false;
    }
    this.faceDetails.visible = false;
    return true;
  }

  resetCamera() {
    this.camera?.position.set(3.15, 2.45, 7.7);
    this.controls?.target.set(0, 1.65, 0);
    this.controls?.update();
  }

  resize() {
    if (!this.renderer || this.disposed) return;
    const { width, height } = this.container.getBoundingClientRect();
    this.visible = width > 0 && height > 0;
    if (!width || !height) return;
    this.camera.aspect = width / height;
    // Portrait/mobile canvases need a slightly wider field of view.
    this.camera.fov = width / height < 0.65 ? 39 : 33;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(width, height, false);
  }

  animatePose(t, delta) {
    delta = clamp(Number.isFinite(delta) ? delta : 0, 0, 0.05);
    const walk = this.motion === 'walk';
    const wave = this.motion === 'wave';
    const turn = this.motion === 'turn';
    const pace = t * 3.6;
    const lerp = 1 - Math.exp(-delta * 9);
    const approach = (object, axis, value) => {
      object.rotation[axis] = THREE.MathUtils.lerp(object.rotation[axis], value, lerp);
    };
    const bob = walk ? Math.abs(Math.sin(pace)) * 0.028 : Math.sin(t * 1.65) * 0.006;
    this.hips.position.y = 1.68 + bob;
    approach(this.hips, 'z', walk ? Math.sin(pace) * 0.035 : Math.sin(t * 0.8) * 0.013);
    approach(this.spine, 'y', walk ? Math.sin(pace) * 0.055 : Math.sin(t * 0.55) * 0.035);
    approach(this.spine, 'z', walk ? -Math.sin(pace) * 0.025 : -0.012);
    approach(this.head, 'y', wave ? -0.12 : Math.sin(t * 0.42) * 0.07);
    approach(this.head, 'z', wave ? -0.05 : Math.sin(t * 0.6) * 0.025);
    if (turn) this.avatar.rotation.y += delta * 0.45;
    else {
      // Return along the shortest route after the 360 degree turntable mode.
      const angle = Math.atan2(Math.sin(this.avatar.rotation.y + 0.06), Math.cos(this.avatar.rotation.y + 0.06));
      this.avatar.rotation.y -= angle * lerp;
    }
    for (const arm of this.arms) {
      const phase = arm.side === 1 ? 0 : Math.PI;
      const waving = wave && arm.side === -1;
      approach(arm.shoulder, 'x', walk ? Math.sin(pace + phase) * 0.32 : waving ? -0.15 : -0.025);
      approach(arm.shoulder, 'z', waving ? -1.1 : arm.side * (0.105 + Math.sin(t * 1.1) * 0.014));
      approach(arm.elbow, 'x', walk ? -0.13 - Math.max(0, Math.sin(pace + phase)) * 0.15 : waving ? -0.18 : -0.10);
      approach(arm.elbow, 'z', waving ? -1.18 - Math.sin(t * 5) * 0.19 : 0);
      approach(arm.hand, 'z', waving ? Math.sin(t * 5) * 0.22 : 0);
    }
    for (const leg of this.legs) {
      const phase = leg.side === 1 ? Math.PI : 0;
      const swing = Math.sin(pace + phase);
      approach(leg.hip, 'x', walk ? swing * 0.32 : leg.side === 1 ? -0.025 : 0.03);
      approach(leg.hip, 'z', leg.side * 0.018);
      approach(leg.knee, 'x', walk ? Math.max(0, -swing) * 0.57 : leg.side === 1 ? 0.025 : 0.045);
      approach(leg.foot, 'x', walk ? Math.max(0, -swing) * -0.17 : 0);
    }
    this.skirt.rotation.x = walk ? Math.sin(pace) * 0.035 : 0;
    this.skirt.rotation.z = walk ? Math.sin(pace) * 0.025 : 0;
    // Ground the planted sole after joint motion and changes to body height.
    // Measuring three sole points also handles toe-off without a sinking heel.
    this.avatar.position.y = 0;
    this.avatar.updateMatrixWorld(true);
    let lowest = Infinity;
    for (const leg of this.legs) {
      for (const z of [-0.035, 0.07, 0.18]) {
        this.footPoint.set(0, -0.092, z).applyMatrix4(leg.foot.matrixWorld);
        lowest = Math.min(lowest, this.footPoint.y);
      }
    }
    this.avatar.position.y = 0.004 - lowest;
  }

  tick(now) {
    if (this.disposed) return;
    const delta = this.lastTime === null ? 0 : clamp((now - this.lastTime) / 1000, 0, 0.05);
    this.lastTime = now;
    if (!this.visible || document.hidden) {
      this.frame = requestAnimationFrame(this.tick);
      return;
    }
    if (!this.paused) {
      this.elapsed += delta;
      this.animatePose(this.elapsed, delta);
    }
    this.controls.update();
    this.renderer.render(this.scene, this.camera);
    this.frame = requestAnimationFrame(this.tick);
  }

  capture() {
    if (!this.renderer) return null;
    this.renderer.render(this.scene, this.camera);
    return this.renderer.domElement.toDataURL('image/png');
  }

  dispose() {
    this.disposed = true;
    cancelAnimationFrame(this.frame);
    this.resizeObserver?.disconnect();
    this.controls?.dispose();
    const geometries = new Set(), materials = new Set(), textures = new Set();
    this.scene?.traverse(object => {
      if (object.geometry) geometries.add(object.geometry);
      if (object.material) {
        const list = Array.isArray(object.material) ? object.material : [object.material];
        for (const material of list) {
          materials.add(material);
          for (const value of Object.values(material)) if (value?.isTexture) textures.add(value);
        }
      }
    });
    for (const texture of textures) texture.dispose();
    for (const material of materials) material.dispose();
    for (const geometry of geometries) geometry.dispose();
    if (this.renderer) {
      this.renderer.domElement.removeEventListener('webglcontextlost', this.contextLost);
      this.renderer.dispose();
      this.renderer.domElement.remove();
    }
    this.fallback?.remove();
  }
}
